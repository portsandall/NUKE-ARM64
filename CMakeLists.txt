cmake_minimum_required(VERSION 3.20)
project(NUKE_ARM64 VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(nuke-platform-probe src/main.c)
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(nuke-platform-probe PRIVATE -Wall -Wextra -Wconversion -Werror)
endif()
# Standalone portability shim test. Does not yet compile the upstream game engine.
include(CTest)
if(BUILD_TESTING)
  find_package(Threads REQUIRED)
  add_executable(nuke-atomic-test tests/atomic_compat.cpp)
  target_include_directories(nuke-atomic-test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
  target_link_libraries(nuke-atomic-test PRIVATE Threads::Threads)
  if(NOT CMAKE_CROSSCOMPILING)
    add_test(NAME physics-atomic-compat COMMAND nuke-atomic-test)
    add_test(NAME nuketown-only COMMAND nuke-platform-probe mp_nuked)
    add_test(NAME reject-other-map COMMAND nuke-platform-probe mp_cracked)
    set_tests_properties(reject-other-map PROPERTIES WILL_FAIL TRUE)
  endif()
endif()

# Optional real upstream translation unit, kept in a separately acquired checkout.
set(NUKE_KISAKBLACK_ROOT "" CACHE PATH "Path to an existing KisakBlack source checkout")
if(NUKE_KISAKBLACK_ROOT)
  set(_nuke_base64_source "${NUKE_KISAKBLACK_ROOT}/src/universal/base64.cpp")
  if(NOT EXISTS "${_nuke_base64_source}")
    message(FATAL_ERROR "KisakBlack Base64 source not found: ${_nuke_base64_source}")
  endif()
  add_library(nuke-upstream-base64 STATIC "${_nuke_base64_source}")
  target_include_directories(nuke-upstream-base64 PUBLIC "${NUKE_KISAKBLACK_ROOT}/src/universal")
  if(BUILD_TESTING)
    add_executable(nuke-base64-smoke tests/base64_smoke.cpp)
    target_include_directories(nuke-base64-smoke PRIVATE "${NUKE_KISAKBLACK_ROOT}/src/universal")
    target_link_libraries(nuke-base64-smoke PRIVATE nuke-upstream-base64)
    if(NOT CMAKE_CROSSCOMPILING)
      add_test(NAME upstream-base64-smoke COMMAND nuke-base64-smoke)
    endif()
  endif()
endif()
